                      )}
                    </div>
                  ))}
                </div>
              )}
            </div>

            {/* Footer avec statistiques */}

                </span>

                  <span style={{ padding: "2px 8px", background: C.white, borderRadius: 6, border: `1px solid ${C.line}`, fontFamily: "monospace", fontSize: 10.5 }}>ESC</span>

                </span>
              </div>

      )}
    </>
